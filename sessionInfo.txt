  suppressWarnings(
    try(
      suppressPackageStartupMessages(library(p, character.only = TRUE)),
      silent = TRUE
    )
  )
}

print(sessionInfo())

cat("\nPackage versions:\n")
for (p in pkgs) {
  v <- tryCatch(as.character(packageVersion(p)), error = function(e) NA_character_)
  cat(p, "\t", v, "\n")
}

sink()
